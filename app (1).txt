def risk_color(score):
    if score < 30:   return "#dc2626"
    elif score < 50: return "#ea580c"
    elif score < 70: return "#ca8a04"
    else:            return "#16a34a"

def risk_label(score):
    if score < 30:   return "🔴 Critical"
    elif score < 50: return "🟠 High"
    elif score < 70: return "🟡 Medium"
    else:            return "🟢 Low"

# ── Dynamic state/sector vocabulary for headline matching ──────────────────────
# Built from your actual data instead of hardcoded, so it stays correct if the
# dataset grows. Sorted longest-first so "Uttar Pradesh" matches before "Pradesh".
ALL_STATES  = sorted(scores["State"].unique(), key=len, reverse=True)
ALL_SECTORS = sorted(scores["Sector"].unique(), key=len, reverse=True)

def detect_affected_nodes(headline: str):
    """
    Scan a headline for any known state and/or sector name.
    Returns a list of matching graph node IDs ('STATE_Sector' format).
    - If both a state AND a sector are found -> exactly one node (that cluster).
    - If only a state is found -> every node in that state.
    - If only a sector is found -> every node in that sector.
    - If neither is found -> empty list.
    """
    text = headline.upper()

    matched_state = next((s for s in ALL_STATES if s.upper() in text), None)
    matched_sector = next((sec for sec in ALL_SECTORS if sec.upper() in text), None)

    if matched_state and matched_sector:
        node_id = f"{matched_state}_{matched_sector}"
        return [node_id] if node_id in G.nodes() else []
    elif matched_state:
        return [n for n in G.nodes() if n.startswith(f"{matched_state}_")]
    elif matched_sector:
        return [n for n in G.nodes() if n.endswith(f"_{matched_sector}")]
    else:
        return []
